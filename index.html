const secret = Math.floor(Math.random() * 100) + 1;
let attempts = 0;

console.log("Я загадал число от 1 до 100. Используй guess(число), чтобы проверить!");

function guess(num) {
  attempts++;
  if (num === secret) {
    console.log(`Угадал! Число было ${secret}. Попыток: ${attempts}`);
  } else if (num > secret) {
    console.log("Моё число меньше.");
  } else if (num < secret) {
    console.log("Моё число больше.");
  } else {
    console.log("Это не число. Попробуй снова.");
  }
}
